  labelfont=bf,
  textfont=bf,
  justification=centering,
  singlelinecheck=true,
  position=bottom,
  skip=8pt
}

}

%======================= HEADING STYLES =======================
% Chapter heading:
% CHAPTER 1: INTRODUCTION
\titleformat{\chapter}[block]
  {\normalfont\bfseries\setstretch{1}\fontsize{16}{19}\selectfont\centering}
  {CHAPTER \thechapter:}
  {0.5em}
  {}


\titlespacing*{\chapter}{0pt}{-16pt}{0pt}

\titleformat{\section}

  {\normalfont\bfseries\fontsize{12}{15}\selectfont}

% Custom \listoffigures
% Wrapped in \AtBeginDocument because tocloft redefines \listoffigures
% at the start of the document and would otherwise override this.
\makeatletter
\AtBeginDocument{%
  \renewcommand{\listoffigures}{%
    \chapter*{\textbf{LIST OF FIGURES}}%
    \addcontentsline{toc}{chapter}{LIST OF FIGURES}%
    \vspace{-15pt}%
    \@starttoc{lof}%
  }%
}
\makeatother
